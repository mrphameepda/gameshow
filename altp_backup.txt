    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.0/firebase-app.js";
    import { getDatabase, ref, push, onValue, query, orderByChild, limitToLast } from "https://www.gstatic.com/firebasejs/10.7.0/firebase-database.js";

    const firebaseConfig = {
        apiKey: "AIzaSyC6hPsvCo_j6bsDWsQZon_2dEWZrBaCvCk",
        authDomain: "bxhtop-85816.firebaseapp.com",
        databaseURL: "https://bxhtop-85816-default-rtdb.asia-southeast1.firebasedatabase.app",
        projectId: "bxhtop-85816",
        appId: "1:393623323872:web:83a299817e2b4f0195bc85"
    };

    const app = initializeApp(firebaseConfig);
    const db = getDatabase(app);

    window.saveGlobal = async (prizeStr) => {
        const scoreVal = parseInt(prizeStr.replace(/[.,]/g, '')) || 0;
        const name = document.getElementById('p-name').value;
        if(!name) return;
        try {
            await push(ref(db, 'leaderboard'), {
                name: name, diocese: document.getElementById('p-diocese').value,
                parish: document.getElementById('p-parish').value,
                prize: prizeStr, score: scoreVal, timestamp: Date.now()
            });
        } catch (e) { console.error(e); }
    };

    window.loadBXH = () => {
        const topScoresQuery = query(ref(db, 'leaderboard'), orderByChild('score'), limitToLast(10));
        onValue(topScoresQuery, (snapshot) => {
            const data = snapshot.val();
            let list = [];
            for (let id in data) list.push(data[id]);
            list.sort((a, b) => b.score - a.score);
            const html = list.map((v, i) => `
                <div class="flex items-center gap-2 p-2 bg-white/5 rounded-lg border border-white/5">
                    <span class="font-black ${i < 3 ? 'text-yellow-500' : 'text-slate-400'}">${i + 1}</span>
                    <div class="flex-1 truncate text-white"><b>${v.name}</b> <br><small class="text-slate-400 uppercase">${v.parish || '...'}</small></div>
                    <div class="text-emerald-400 font-black text-xs">${v.prize}</div>
                </div>`).join("");
            document.getElementById('modal-list').innerHTML = html;
            if(document.getElementById('side-bxh')) document.getElementById('side-bxh').innerHTML = html;
        });
    };
    window.loadBXH();
</script>
